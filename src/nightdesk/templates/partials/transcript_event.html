{# Per-event renderers for canonical transcript events. Each macro renders one event type as a self-contained DOM block. The JS live-tail in templates/partials/transcript_panel.html mirrors these shapes when appending events client-side. Keep the two sides in sync — same DOM, same CSS classes (defined in static/app.css under .tc-card / .tool-result / .thinking-block / .assistant-text / .diff). Pure helpers come from nightdesk.transcript_view, exposed as the Jinja global `tv` in nightdesk.api.app.create_app. #} {% import "partials/tool_card.html" as tc %} {% macro evt_meta(evt) %}
run {{ (evt.get('run_id') or '')[:8] }} started {{ evt.get('ts','') }}
{% endmacro %} {# assistant_text: first-class block (left accent border, elevated bg, author chip) so the model's prose stands out from surrounding tool cards. Fenced-code blocks render as 
 so newlines
   survive untouched.

   Empty-text events (some SDK variants emit one when only tool_use
   blocks were produced) render nothing — otherwise we'd show a bare
   "claude" chip with no body. Mirrors the empty-text guard in the JS
   live-tail (templates/partials/transcript_panel.html).

   The author chip prefers ``evt.model`` (if a future translator carries
   it through) and falls back to 'claude'. The meta event currently
   doesn't surface model into per-assistant events, so the fallback is
   what fires today. #}

  {% set text = evt.get('text','') %}
  {% if text and text.strip() %}
    

      
{{ evt.get('model') or 'claude' }}

      {% for seg in tv.assistant_segments(text) %}
        {% if seg.kind == 'code' %}
          
{{ seg.body }}

        {% elif seg.body %}
          
{{ seg.body }}

        {% endif %}
      {% endfor %}
    

  {% endif %}
{% endmacro %}

{# thinking: italic muted, left-border, collapsed behind 'thinking (N tokens)'.
   N tokens is a whitespace-token approximation since the canonical event
   carries no token count.

   Some Claude variants return thinking blocks with empty content (the raw
   chain-of-thought never leaves the API). The translator drops those
   going forward, but legacy transcripts on disk still contain them — so
   render nothing when the text is empty/whitespace. #}
{% macro evt_thinking(evt) %}
  {% set text = evt.get('text','') %}
  {% if text and text.strip() %}
    {% set tokens = text.split() | length %}
    

      thinking ({{ tokens }} tokens)
      
{{ text }}

    

  {% endif %}
{% endmacro %}

{# evt_result: terminal result card. For SUCCESSFUL runs this renders the green
   "done" card. For FAILED runs (subtype != success) it renders nothing — the
   failure is already surfaced by the always-visible top banner (see
   transcript_panel.html) and, when present, the richer evt_worker_error card
   (which carries kind + stacktrace). This avoids the double red card at the
   bottom of a failed transcript. #}
{% macro evt_result(evt) %}
  {% set subtype = evt.get('subtype','success') %}
  {% if subtype == 'success' %}
  

    
done

    
{{ evt.get('summary','') }}

  

  {% endif %}
{% endmacro %}

{# worker_error: nightdesk-level failure surfaced as the final event of a

{# rate_limit: the agent SDK reported a rate-limit status change. Render an

    {% if raw %}
      

        raw response
        
{{ raw }}

      

    {% endif %}
    {% if resets_at %}
      

        next request in

    {% endif %}

{# subagent: a sub-agent (Task tool) lifecycle card. The run's agent spawned a
   sub-agent (e.g. Explore); the SDK streams started/progress/notification
   updates that the renderer collapses (per task_id) into this single updating
   card. NOT a worker_error — a healthy run produces these. The card carries
   ``data-subagent-task-id`` so the live-tail JS can update the same node in
   place when later phases arrive. The JS live-tail in transcript_panel.html
   mirrors this shape. #}
{% macro evt_subagent(evt) %}
  {% set s = tv.subagent_summary(evt) %}
  {% set tid = evt.get('task_id') or evt.get('tool_use_id') or '' %}
  

    

      
        subagent
        {{ s.label }}
        
          {% if s.done %}{% if s.failed %}failed{% else %}completed{% endif %}{% elif s.detail %}{{ s.detail }}{% else %}running{% endif %}
        
      
      

        {% if s.detail and not s.done %}
{{ s.detail }}
{% endif %}
        

          {% if s.tool_uses %}{{ s.tool_uses }} tool use{{ '' if s.tool_uses == 1 else 's' }}{% endif %}
          {% if s.duration %}{{ s.duration }}{% endif %}
          {% if s.tokens %}{{ '{:,}'.format(s.tokens) }} tokens{% endif %}
        

        {% if s.summary %}
{{ s.summary }}
{% endif %}
      

    

  

{% endmacro %}

{% macro render_event(evt, paired_result=None) %}
  {% set t = evt.get('type','') %}
  {% if t == 'meta' %}{{ evt_meta(evt) }}
  {% elif t == 'assistant_text' %}{{ evt_assistant_text(evt) }}
  {% elif t == 'thinking' %}{{ evt_thinking(evt) }}
  {% elif t == 'tool_use' %}{{ evt_tool_use(evt, paired_result) }}
  {% elif t == 'tool_result' %}{{ evt_tool_result(evt) }}
  {% elif t == 'result' %}{{ evt_result(evt) }}
  {% elif t == 'worker_error' %}{{ evt_worker_error(evt) }}
  {% elif t == 'rate_limit' %}{{ evt_rate_limit(evt) }}
  {% elif t == 'cancelled' %}{{ evt_cancelled(evt) }}
  {% elif t == 'subagent' %}{{ evt_subagent(evt) }}
  {% endif %}
{% endmacro %}

{# Render a GroupedEvent node (from tv.group_by_subagent): the event itself,
   plus — for a sub-agent — its nested child tool calls. The child wrapper
   carries data-parent-tool-use-id so the sidebar can filter by sub-agent.
   The JS live-tail in transcript_panel.html mirrors this nesting. #}
{% macro render_grouped(node) %}
  {% set t = node.event.get('type','') %}
  {% if t == 'subagent' %}
    

      {{ evt_subagent(node.event) }}
      {% if node.children %}
      

        {{ node.children|length }} tool call{{ '' if node.children|length == 1 else 's' }}
        

          {% for child in node.children %}
            

              {{ render_event(child.event, child.paired_result) }}
            

          {% endfor %}
        

      

      {% endif %}
    

  {% else %}
    {{ render_event(node.event, node.paired_result) }}
  {% endif %}
{% endmacro %}
